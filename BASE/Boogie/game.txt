  }
}

function crearMatrizVacia(tamanio) {
  const matriz = [];
  for (let fila = 0; fila < tamanio; fila++) {
    matriz[fila] = [];
    for (let col = 0; col < tamanio; col++) {
      matriz[fila][col] = null;
    }
  }
  return matriz; //
}

function elegirLetra(opciones) {
  const i = Math.floor(Math.random() * opciones.length);
  return opciones[i]; //opciones es una matriz, y opciones[i] es un letra
}

function validarCondiciones(matrizB, fila, columna, letra) {
  //no repetir con el de izquierda
  if (fila > 0 && matrizB[fila][columna - 1] == letra) {
    return false;
  }
  //no repetir con el de arriba
  if (fila > 0 && matrizB[fila - 1][columna] == letra) {
    return false;
  }
  //no repetir con el de la derecha - es al pedo.
  // if (fila > 0 && matrizB[fila][columna + 1] == letra) {
  //   return false;
  // }
  return true; //si paso los check entonces es true.
}
function rellenarMatriz(tamanio) {
  // Tiene 3 helper functions, definidas arriba.
  const matrizA = crearMatrizVacia(tamanio); // crearMatrizVacia te devuelve una matriz,que la titulo matrizA
  //la recorres rellenandola.
  for (let fil = 0; fil < matrizA.length; fil++) {
    for (let col = 0; col < matrizA.length; col++) {
      let letra = null;
      const MAX_INTENTOS = 50; // para encontrar el error
      for (let intentos = 0; intentos < MAX_INTENTOS; intentos++) {
        //Primer IF - vocales en la diagonal
        if (fil === col) {
          letra = elegirLetra(VOWELS);
        } else {
          letra = elegirLetra(ALL_LETTERS);
        }
        //segundo IF - VALIDACION DE LA DERECHA, IZQ, ARRI QUE NO SE REPITA.
        if (validarCondiciones(matrizA, fil, col, letra)) {
          break; // salimos → letra válida
        } else {
          letra = null;
        }
      }
      if (letra == null) {
        throw new Error(`No se encontro letra valida en ${fil} y ${col}`);
      }
      matrizA[fil][col] = letra;
    }
  }
  return matrizA;
}

function renderizaBoard(matrizB) {

//FUNCIONES PARA QUE VIBREN LOS ULTIMOS 10 SEGUNDOS
function startLetterVibration() {
  const letters = document.querySelectorAll(".cell-letter");
  letters.forEach((cell) => cell.classList.add("vibrate"));
}

function stopLetterVibration() {
  const letters = document.querySelectorAll(".cell-letter.vibrate");
  letters.forEach((cell) => cell.classList.remove("vibrate"));
}

//CUENTA REGRESIVA EN EL JUEGO CON LA SOPA DE LETRAS
function startCountdown() {
  let countdownTime = opcionElegidaActualizada.Duracion * 60; // segundos totales
  const countdownText = document.getElementById("cuentaregres2");
  const circleProgress = document.querySelector(".circle-progress");

  const circleLength = 283;

  let timeLeft = countdownTime;
  let intervalTime = 1000; // 1 segundo
  let hasStartedVibration = false; // 👈 nuevo flag

  // intervalo normal
  var interval = setInterval(function updateCountdown() {
    timeLeft--;
    countdownText.textContent = timeLeft;

    const offset = circleLength - circleLength * (timeLeft / countdownTime);
    circleProgress.style.strokeDashoffset = offset;

    // 👉 activar vibración en los últimos 10 segundos
    if (!hasStartedVibration && timeLeft <= 10 && timeLeft > 0) {
      startLetterVibration();
      hasStartedVibration = true;
    }

    // acelerar en los últimos 10 segundos (2x speed)
    if (timeLeft <= 10 && intervalTime !== 500) {
      intervalTime = 500;
      clearInterval(interval);
      startAcceleratedCountdown();
    }

    if (timeLeft <= 0) {
      clearInterval(interval);
      stopLetterVibration(); // 👈 apagar vibración
      countdownText.textContent = "¡Tiempo!";
      circleProgress.style.stroke = "#4CAF50";
      gridBoard2.classList.add("screen-hidden");
      showRandomImage();
    }
  }, intervalTime);

  // intervalo acelerado
  function startAcceleratedCountdown() {
    var acceleratedInterval = setInterval(
      function updateAcceleratedCountdown() {
        timeLeft--;
        countdownText.textContent = timeLeft;

        const offset = circleLength - circleLength * (timeLeft / countdownTime);
        circleProgress.style.strokeDashoffset = offset;

        // por si la aceleración arranca justo en 10
        if (!hasStartedVibration && timeLeft <= 10 && timeLeft > 0) {
          startLetterVibration();
          hasStartedVibration = true;
        }

        if (timeLeft <= 0) {
          clearInterval(acceleratedInterval);
          stopLetterVibration(); // 👈 apagar vibración
          countdownText.textContent = "¡Tiempo!";
          countdownText.style.fontSize = "1rem";
          circleProgress.style.stroke = "#4CAF50";
          gridBoard2.classList.add("screen-hidden");
          showRandomImage();
        }
      },
      intervalTime
    ); // 500ms
  }
}

function showRandomImage() {
  // Seleccionar una imagen aleatoria de la lista
  const randomIndex = Math.floor(Math.random() * imageUrls.length);
  const randomImageUrl = imageUrls[randomIndex];

  // checkImagePermissions(randomImageUrl); // Comprobar si la imagen tiene permisos adecuados antes de cargarla - ESTA ABAJO, por el tema de proxy.

  // Crear un elemento <img> para mostrar la imagen
  const imageElement = document.createElement("img");
  imageElement.src = randomImageUrl;
  imageElement.alt = "Imagen aleatoria";
  imageElement.style.maxWidth = "100%"; // Ajusta el tamaño según lo necesites
  imageElement.style.height = "auto"; // Mantiene la proporción de la imagen

  // Verificar si la imagen se carga correctamente
  imageElement.onerror = function () {
    console.log("Error al cargar la imagen: " + randomImageUrl);
  };
  // Mostrar la imagen en la pantalla (por ejemplo, en el contenedor de "screen-end")
  const screenEnd = document.getElementById("screen-end");

  // Cambiar el fondo de la pantalla a negro
  screenEnd.style.backgroundColor = "black"; // Establece el fondo negro
  screenEnd.style.display = "flex"; // Usamos flexbox para centrar la imagen
  screenEnd.style.alignItems = "center"; // Centrar verticalmente
  screenEnd.style.justifyContent = "center"; // Centrar horizontalmente
  screenEnd.style.height = "100vh"; // Asegura que la pantalla ocupe toda la altura de la ventana
  screenEnd.classList.remove("screen-hidden"); // Asegúrate de que la pantalla de final del juego sea visible
  screenEnd.appendChild(imageElement); // Añadir la imagen a la pantalla
}
