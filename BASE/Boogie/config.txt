// selecciono todos los botones de un div.“Buscá todos los elementos con clase .chip que estén dentro de un elemento .choice.”
const Cantletras = document.querySelectorAll(".choicetam .chip");
const Durac = document.querySelectorAll(".choicedura .chip");

//Desenmarcar botones de la duracion o del tamaño.
function desenmarcarBotones(y) {
  y.forEach((boton) => boton.classList.remove("seleccionado"));
}

Cantletras.forEach((boton) => {
  boton.addEventListener("click", () => {
    desenmarcarBotones(Cantletras); //desenmarcas todos los botones
    opcElegida.Qletras = parseInt(boton.textContent); //actualizamos el objeto
    boton.classList.add("seleccionado"); // cambiamos el foramto
    updateHint(opcElegida.Duracion, opcElegida.Qletras); //actualizar el hint.
  });
});
